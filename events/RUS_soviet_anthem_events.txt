add_namespace = RUS_soviet_anthem

# The call for a new anthem opens as soon as the new foreign-policy focus is selected.
country_event = {
	id = RUS_soviet_anthem.1
	title = RUS_soviet_anthem.1.t
	desc = RUS_soviet_anthem.1.d
	is_triggered_only = yes
	trigger = {
		original_tag = RUS
		NOT = { has_country_flag = RUS_soviet_anthem_contest_open }
		NOT = { has_country_flag = RUS_soviet_anthem_decision_offered }
	}
	immediate = {
		set_country_flag = RUS_soviet_anthem_contest_open
	}
	option = {
		name = RUS_soviet_anthem.1.a
	}
}
