				type = annex_everything
			}
		}

		hidden_effect = {
			set_country_flag = RUS_future_foreign_062_claims_only
			complete_national_focus = RUS_future_foreign_062
		}
	}

# Resolve readiness once, when the delayed event reaches the queue.
country_event = {
	id = RUS_future_foreign_policy_events.7
	hidden = yes
	is_triggered_only = yes
	trigger = {
		focus_progress = { focus = RUS_future_foreign_062 progress > 0 }
		NOT = { has_completed_focus = RUS_future_foreign_062 }
		NOT = { has_country_flag = RUS_future_foreign_062_readiness_resolved }
	}
	immediate = {
		set_country_flag = RUS_future_foreign_062_readiness_resolved
		if = {
			limit = { power_balance_value = { id = RUS_europe_intervention_power_balance value < 0.75 } }
			country_event = { id = RUS_future_foreign_policy_events.5 }
		}
		else = {
			country_event = { id = RUS_future_foreign_policy_events.6 }
		}
	}
}
