country_event = {
	id = rus_stalin_white_revolt_events.4
	hidden = yes
	is_triggered_only = yes
	trigger = {
		has_country_flag = RUS_white_revolt_active
		NOT = { has_country_flag = RUS_white_revolt_counterattack_active }
	}
	immediate = {
		if = {
			limit = {
				any_state = {
					has_state_flag = RUS_white_revolt_red_control
					OR = {
						AND = {
							OR = { state = 208 state = 216 state = 244 }
							ROOT = { NOT = { has_country_flag = RUS_white_revolt_petrograd_won } }
						}
						AND = {
							OR = { state = 245 state = 265 state = 401 state = 238 }
							ROOT = { NOT = { has_country_flag = RUS_white_revolt_tsaritsyn_won } }
						}
					}
				}
			}
			random_state = {
				limit = {
					has_state_flag = RUS_white_revolt_red_control
					OR = {
						AND = {
							OR = { state = 208 state = 216 state = 244 }
							ROOT = { NOT = { has_country_flag = RUS_white_revolt_petrograd_won } }
						}
						AND = {
							OR = { state = 245 state = 265 state = 401 state = 238 }
							ROOT = { NOT = { has_country_flag = RUS_white_revolt_tsaritsyn_won } }
						}
					}
				}
				save_event_target_as = RUS_white_revolt_counterattack_state
				set_state_flag = RUS_white_revolt_under_attack
			}
			set_country_flag = RUS_white_revolt_counterattack_active
			activate_mission = RUS_white_revolt_counterattack_mission
			country_event = rus_stalin_white_revolt_events.10
		}
		else = {
			country_event = { id = rus_stalin_white_revolt_events.4 days = 7 }
		}
	}
}
