unique_id = RUS_stalin_adapted_achievements

RUS_stalin_adapted_steel_comrade_achievement = {
	possible = {
		tag = RUS
	}

	happened = {
		if = {
			limit = {
				NOT = { has_global_flag = disallow_achievements_flag }
				OR = {
					NOT = { has_global_flag = on_startup_completed }
					is_ai = no
				}
			}
			custom_override_tooltip = {
				tooltip = RUS_stalin_adapted_steel_comrade_achievement_tt
				has_country_flag = RUS_stalin_communist_path
				has_completed_focus = RUS_unite_under_stalin
				has_country_leader = { character = RUS_iosif_stalin }
			}
		}
		else_if = {
			limit = { has_global_flag = disallow_achievements_flag }
			custom_override_tooltip = {
				tooltip = achievements_disabled
				always = no
			}
		}
		else = {
			custom_override_tooltip = {
				tooltip = achievements_disabled_observer
				always = no
			}
		}
	}
}
