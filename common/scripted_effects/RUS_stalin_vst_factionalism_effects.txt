RUS_stalin_calculate_vst_factionalism_relationship_effects = {
	set_variable = { RUS_vst_factionalism_extra_pp_gain = 0 }
	set_variable = { RUS_vst_factionalism_weekly_stability = 0 }
	set_variable = { RUS_vst_factionalism_building_slots_factor = 0 }
	set_variable = { RUS_vst_factionalism_arms_factory_cost_factor = 0 }
	set_variable = { RUS_vst_factionalism_monthly_cic = 0 }

	if = {
		limit = { check_variable = { vst_centre_var < 4 } }
		set_variable = { RUS_vst_factionalism_internationalist_stability = 4 }
		subtract_from_variable = { RUS_vst_factionalism_internationalist_stability = vst_centre_var }
		multiply_variable = { RUS_vst_factionalism_internationalist_stability = -0.001 }
		add_to_variable = { RUS_vst_factionalism_weekly_stability = RUS_vst_factionalism_internationalist_stability }
	}
	else = {
		set_variable = { RUS_vst_factionalism_monthly_cic = vst_centre_var }
		subtract_from_variable = { RUS_vst_factionalism_monthly_cic = 4 }
		multiply_variable = { RUS_vst_factionalism_monthly_cic = 5000 }
	}

	if = {
		limit = { check_variable = { sr_var < 4 } }
		set_variable = { RUS_vst_factionalism_building_slots_factor = 4 }
		subtract_from_variable = { RUS_vst_factionalism_building_slots_factor = sr_var }
		multiply_variable = { RUS_vst_factionalism_building_slots_factor = -0.10 }
	}
	else = {
		set_variable = { RUS_vst_factionalism_sr_stability = sr_var }
		subtract_from_variable = { RUS_vst_factionalism_sr_stability = 4 }
		multiply_variable = { RUS_vst_factionalism_sr_stability = 0.001 }
		add_to_variable = { RUS_vst_factionalism_weekly_stability = RUS_vst_factionalism_sr_stability }
	}

	if = {
		limit = { check_variable = { rkp_var < 4 } }
		set_variable = { RUS_vst_factionalism_arms_factory_cost_factor = 4 }
		subtract_from_variable = { RUS_vst_factionalism_arms_factory_cost_factor = rkp_var }
		multiply_variable = { RUS_vst_factionalism_arms_factory_cost_factor = 0.10 }
	}
	else = {
		set_variable = { RUS_vst_factionalism_extra_pp_gain = rkp_var }
		subtract_from_variable = { RUS_vst_factionalism_extra_pp_gain = 4 }
		divide_variable = { RUS_vst_factionalism_extra_pp_gain = 25 }
	}
}

RUS_stalin_refresh_vst_factionalism_dynamic_modifier = {
	if = {
		limit = {
			OR = {
				has_idea = RUS_vst_party_unity_idea
				has_idea = RUS_vst_party_factionalism_idea
			}
		}
		if = {
			limit = {
				has_country_flag = RUS_stalin_communist_path
				NOT = { has_country_flag = RUS_stalin_vst_factionalism_effects_frozen }
			}
			RUS_stalin_freeze_vst_factionalism_relationship_effects = yes
		}
		else_if = {
			limit = { NOT = { has_country_flag = RUS_stalin_vst_factionalism_effects_frozen } }
			RUS_stalin_calculate_vst_factionalism_relationship_effects = yes
		}

		set_variable = { RUS_vst_factionalism_total_pp_gain = vst_popularity_var }
		add_to_variable = { RUS_vst_factionalism_total_pp_gain = RUS_vst_factionalism_extra_pp_gain }
		force_update_dynamic_modifier = yes
	}
}

RUS_stalin_freeze_vst_factionalism_relationship_effects = {
	if = {
		limit = { NOT = { has_country_flag = RUS_stalin_vst_factionalism_effects_frozen } }
		RUS_stalin_calculate_vst_factionalism_relationship_effects = yes
		set_country_flag = RUS_stalin_vst_factionalism_effects_frozen
	}
	set_variable = { RUS_vst_factionalism_total_pp_gain = vst_popularity_var }
	add_to_variable = { RUS_vst_factionalism_total_pp_gain = RUS_vst_factionalism_extra_pp_gain }
	force_update_dynamic_modifier = yes
}

RUS_stalin_apply_vst_factionalism_monthly_surplus = {
	if = {
		limit = {
			OR = {
				has_idea = RUS_vst_party_unity_idea
				has_idea = RUS_vst_party_factionalism_idea
			}
			check_variable = { RUS_vst_factionalism_monthly_cic > 0 }
		}
		add_cic = RUS_vst_factionalism_monthly_cic
	}
}
