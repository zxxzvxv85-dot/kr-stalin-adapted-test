RUS_stalin_convert_totalist_support_to_communist = {
	if = {
		limit = { totalist > 0 }
		set_temp_variable = { RUS_stalin_totalist_popularity = party_popularity@totalist }
		add_popularity = {
			ideology = communist
			popularity = RUS_stalin_totalist_popularity
		}
		multiply_temp_variable = { RUS_stalin_totalist_popularity = -1 }
		add_popularity = {
			ideology = totalist
			popularity = RUS_stalin_totalist_popularity
		}
	}
}

RUS_stalin_set_subject_radical_socialist_if_communist_path = {
	if = {
		limit = {
			RUS = {
				OR = {
					has_country_flag = RUS_stalin_communist_path
					has_government = communist
				}
			}
		}
		RUS_stalin_set_country_radical_socialist_for_subject_release = yes
	}
}

RUS_stalin_set_released_country_radical_socialist_if_from_communist_russia = {
	if = {
		limit = {
			NOT = { tag = RUS }
			FROM = {
				original_tag = RUS
				OR = {
					has_country_flag = RUS_stalin_communist_path
					has_government = communist
				}
			}
		}
		RUS_stalin_set_country_radical_socialist_for_subject_release = yes
		country_event = { id = stalin_form_sov.20 hours = 1 }
	}
}
