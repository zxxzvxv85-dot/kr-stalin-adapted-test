# Country scope. Create Budyonny directly as a field marshal so old saves do not depend on a new character token.
RUS_generate_semyon_budyonny = {
	if = {
		limit = { NOT = { has_country_flag = RUS_fr_budyonny_field_marshal_created } }
		create_field_marshal = {
			name = "谢苗·布琼尼"
			gfx = "GFX_portrait_SOV_semyon_budyonny"
			traits = { old_guard inflexible_strategist politically_connected }
			skill = 1
			attack_skill = 1
			defense_skill = 1
			planning_skill = 1
			logistics_skill = 1
		}
		set_country_flag = RUS_fr_budyonny_field_marshal_created
		set_country_flag = RUS_fr_budyonny_recruited
	}
}

# Country scope. Recruit or repair the separate high-command character for both new and old saves.
RUS_generate_semyon_budyonny_air_assault_chief = {
	if = {
		limit = { NOT = { has_character = RUS_semyon_budyonny_air_assault_chief } }
		log = "[GetDateText]: [ROOT.GetName] transfers the static Budyonny air-assault advisor into the country"
		set_nationality = {
			character = RUS_semyon_budyonny_air_assault_chief
			target_country = ROOT
		}
	}
	if = {
		limit = {
			has_character = RUS_semyon_budyonny_air_assault_chief
			RUS_semyon_budyonny_air_assault_chief = { has_advisor_role = army_chief }
		}
		log = "[GetDateText]: [ROOT.GetName] removes Budyonny's obsolete army-chief role"
		every_character = {
			limit = { is_character = RUS_semyon_budyonny_air_assault_chief }
			remove_advisor_role = { slot = army_chief }
		}
	}
	if = {
		limit = {
			has_character = RUS_semyon_budyonny_air_assault_chief
			RUS_semyon_budyonny_air_assault_chief = { NOT = { has_advisor_role = high_command } }
		}
		log = "[GetDateText]: [ROOT.GetName] repairs Budyonny's missing high-command role"
		add_advisor_role = {
			character = RUS_semyon_budyonny_air_assault_chief
			advisor = {
				slot = high_command
				idea_token = RUS_semyon_budyonny_air_assault_high_command
				ledger = army
				traits = { RUS_budyonny_air_assault_chief }
				cost = 50
				ai_will_do = { base = 100 }
			}
			activate = yes
		}
	}
	if = {
		limit = { has_character = RUS_semyon_budyonny_air_assault_chief }
		set_country_flag = RUS_semyon_budyonny_air_assault_chief_present
	}
}
