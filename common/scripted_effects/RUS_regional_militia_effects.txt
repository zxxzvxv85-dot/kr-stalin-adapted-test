# Spawn limited local volunteer formations on the Russian side of the border.

RUS_rd_GEO_form_militia = {
 if = {
  limit = {
   check_variable = { RUS_rd_GEO_militia_raised < 2 }
   OR = {
    AND = { NOT = { has_war_with = GEO } any_owned_state = { is_controlled_by = ROOT impassable = no any_neighbor_state = { is_controlled_by = GEO } } }
    AND = { has_war_with = GEO GEO = { any_controlled_state = { impassable = no any_neighbor_state = { is_controlled_by = ROOT } } } }
   }
  }
  if = {
   limit = { NOT = { has_template = "格鲁吉亚志愿民兵" } }
   division_template = {
    name = "格鲁吉亚志愿民兵"
    is_locked = yes
    regiments = {
     militia = { x = 0 y = 0 }
     militia = { x = 0 y = 1 }
     militia = { x = 1 y = 0 }
     militia = { x = 1 y = 1 }
    }
   }
  }
  if = {
   limit = { has_war_with = GEO }
   GEO = {
    random_controlled_state = {
     limit = { impassable = no any_neighbor_state = { is_controlled_by = ROOT } }
     create_unit = {
      division = "division_template = \"格鲁吉亚志愿民兵\" start_experience_factor = 0.1 start_equipment_factor = 1.0"
      owner = ROOT
      allow_spawning_on_enemy_provs = yes
      country_score = { base = 0 modifier = { tag = GEO add = 1000 } }
      count = 1
     }
    }
   }
  }
  else = {
  random_owned_controlled_state = {
   limit = { impassable = no any_neighbor_state = { is_controlled_by = GEO } }
   create_unit = {
    division = "division_template = \"格鲁吉亚志愿民兵\" start_experience_factor = 0.1 start_equipment_factor = 1.0"
    owner = PREV
    allow_spawning_on_enemy_provs = no
    count = 1
   }
  }
  }
  add_to_variable = { RUS_rd_GEO_militia_raised = 1 }
 }
}
