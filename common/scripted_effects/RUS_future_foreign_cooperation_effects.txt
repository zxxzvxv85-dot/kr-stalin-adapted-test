RUS_future_foreign_joint_deterrence = {
	if = {
		limit = {
			FRA = {
				exists = yes
				has_socialist_government = yes
				NOT = { has_war_with = ROOT }
			}
		}
		give_guarantee = FRA
		add_opinion_modifier = { target = FRA modifier = RUS_future_foreign_joint_deterrence_opinion }
		FRA = {
			give_guarantee = ROOT
			add_opinion_modifier = { target = ROOT modifier = RUS_future_foreign_joint_deterrence_opinion }
		}
		if = {
			limit = {
				has_war_with = GER
				FRA = { has_war_with = GER }
			}
			add_timed_idea = { idea = RUS_franco_russian_joint_deterrence days = 365 }
			FRA = { add_timed_idea = { idea = RUS_franco_russian_joint_deterrence days = 365 } }
		}
		else = {
			add_timed_idea = { idea = RUS_franco_russian_joint_deterrence days = 180 }
			FRA = { add_timed_idea = { idea = RUS_franco_russian_joint_deterrence days = 180 } }
		}
	}
}

RUS_future_foreign_refresh_red_flags = {
	set_variable = { RUS_soviets_and_phalanstere_political_power = 0 }
	set_variable = { RUS_soviets_and_phalanstere_relations_cost = 0 }
	set_variable = { RUS_soviets_and_phalanstere_volunteers = 0 }
	if = {
		limit = { has_completed_focus = RUS_future_foreign_056 }
		set_variable = { RUS_soviets_and_phalanstere_political_power = 0.05 }
		set_variable = { RUS_soviets_and_phalanstere_relations_cost = -0.25 }
		set_variable = { RUS_soviets_and_phalanstere_volunteers = 2 }
	}
}

RUS_future_foreign_pay_aid_pp = {
	if = {
		limit = { RUS_future_foreign_has_balkan_aid_discount = yes }
		RUS_diplomacy_pay_15 = yes
	}
	else_if = {
		limit = { has_completed_focus = RUS_future_foreign_056 }
		RUS_diplomacy_pay_24 = yes
	}
	else = { RUS_diplomacy_pay_30 = yes }
}

RUS_future_foreign_pay_military_aid_materials = {
	if = {
		limit = { RUS_future_foreign_has_balkan_aid_discount = yes }
		add_to_variable = { RUS_fr_military_reform_materials = -10 }
	}
	else = { add_to_variable = { RUS_fr_military_reform_materials = -20 } }
	RUS_fr_refresh_military_reform_materials = yes
}

RUS_future_foreign_pay_civil_aid_score = {
	if = {
		limit = { RUS_future_foreign_has_balkan_aid_discount = yes }
		add_to_variable = { RUS_first_five_year_plan_score = -0.5 }
	}
	else = { add_to_variable = { RUS_first_five_year_plan_score = -1 } }
	RUS_stalin_refresh_first_five_year_plan_stage = yes
}
