RUS_education_reform_start = {
	if = {
		limit = { NOT = { has_country_flag = RUS_education_reform_started } }
		set_country_flag = RUS_education_reform_started
		set_country_flag = RUS_education_reform_baseline_applied
		set_country_flag = RUS_education_recruitable_population_v1_migrated
		clr_country_flag = RUS_education_reform_completed
		remove_ideas = RUS_malformed_education_system
		set_variable = { RUS_education_reform_active_decisions = 0 }
		set_variable = { RUS_education_reform_repeat_count = 0 }
		set_variable = { RUS_education_reform_repeat_limit = 18 }
		set_variable = { RUS_kalinin_education_time_75 = 75 }
		set_variable = { RUS_kalinin_education_time_180 = 180 }
		set_variable = { RUS_kalinin_education_penalty_per_decision = -0.01 }
		set_variable = { RUS_education_research_speed = -0.05 }
		set_variable = { RUS_education_start_efficiency = -0.10 }
		set_variable = { RUS_education_efficiency_growth = -0.10 }
		set_variable = { RUS_education_consumer_goods_factor = 0 }
		set_variable = { RUS_education_political_power_factor = 0 }
		set_variable = { RUS_education_political_power_gain = 0 }
		set_variable = { RUS_education_mio_funds_gain = 0 }
		set_variable = { RUS_education_mio_policy_cost = 0 }
		set_variable = { RUS_education_party_popularity_stability_factor = 0 }
		set_variable = { RUS_education_recruitable_population_factor = 0 }
		hidden_effect = {
			if = {
				limit = { has_dynamic_modifier = { modifier = RUS_completed_education_reform } }
				remove_dynamic_modifier = { modifier = RUS_completed_education_reform }
			}
			if = {
				limit = { has_dynamic_modifier = { modifier = RUS_reforming_education_system } }
				remove_dynamic_modifier = { modifier = RUS_reforming_education_system }
			}
		}
		add_dynamic_modifier = { modifier = RUS_reforming_education_system }
	}
	if = {
		limit = {
			has_country_flag = RUS_education_reform_started
			NOT = { has_country_flag = RUS_education_reform_completed }
			has_idea = RUS_malformed_education_system
		}
		remove_ideas = RUS_malformed_education_system
	}
	if = {
		limit = {
			has_country_flag = RUS_education_reform_started
			NOT = { has_country_flag = RUS_education_reform_completed }
			NOT = { has_dynamic_modifier = { modifier = RUS_reforming_education_system } }
		}
		add_dynamic_modifier = { modifier = RUS_reforming_education_system }
	}
}

RUS_education_reform_start_decision = {
	add_to_variable = { RUS_education_reform_active_decisions = 1 }
}

RUS_education_reform_ensure_new_culture_variables = {
	if = {
		limit = { NOT = { has_variable = RUS_education_consumer_goods_factor } }
		set_variable = { RUS_education_consumer_goods_factor = 0 }
	}
	if = {
		limit = { NOT = { has_variable = RUS_education_political_power_factor } }
		set_variable = { RUS_education_political_power_factor = 0 }
	}
	if = {
		limit = { NOT = { has_variable = RUS_education_research_speed } }
		set_variable = { RUS_education_research_speed = 0 }
	}
	if = {
		limit = { NOT = { has_variable = RUS_education_party_popularity_stability_factor } }
		set_variable = { RUS_education_party_popularity_stability_factor = 0 }
	}
	if = {
		limit = {
			NOT = { has_country_flag = RUS_education_reform_started }
			NOT = { has_country_flag = RUS_education_reform_completed }
		}
		RUS_education_reform_start = yes
	}
}

RUS_education_reform_finish_decision = {
	if = {
		limit = { check_variable = { RUS_education_reform_active_decisions > 0 } }
		subtract_from_variable = { RUS_education_reform_active_decisions = 1 }
	}
}

RUS_education_reform_complete_decision = {
	RUS_education_reform_finish_decision = yes
	custom_effect_tooltip = RUS_education_reform_recruitable_population_dynamic_penalty_tt
	hidden_effect = {
		if = {
			limit = { NOT = { has_variable = RUS_education_recruitable_population_factor } }
			set_variable = { RUS_education_recruitable_population_factor = 0 }
		}
		add_to_variable = { RUS_education_recruitable_population_factor = -0.01 }
		force_update_dynamic_modifier = yes
	}
}

RUS_education_reform_add_repeat = {
	add_to_variable = { RUS_education_reform_repeat_count = 1 }
}

RUS_education_reform_check_completion = {
	if = {
		limit = {
			NOT = { has_country_flag = RUS_education_reform_completed }
			NOT = { check_variable = { RUS_education_reform_repeat_count < RUS_education_reform_repeat_limit } }
			has_country_flag = RUS_edu_compulsory_education_done
			has_country_flag = RUS_edu_higher_education_done
			has_country_flag = RUS_edu_grassroots_cadres_done
			has_country_flag = RUS_edu_cadre_rotation_done
			has_country_flag = RUS_edu_cultural_propaganda_done
			has_country_flag = RUS_edu_production_tech_done
		}
		set_country_flag = RUS_education_reform_completed
		clr_country_flag = RUS_education_reform_started
		if = {
			limit = { has_dynamic_modifier = { modifier = RUS_reforming_education_system } }
			remove_dynamic_modifier = { modifier = RUS_reforming_education_system }
		}
		add_dynamic_modifier = { modifier = RUS_completed_education_reform }
	}
}
