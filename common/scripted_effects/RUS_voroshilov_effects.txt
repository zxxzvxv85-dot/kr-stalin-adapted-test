# Country scope. Voroshilov is recruited at game start without a command role;
# completing the relevant military reform grants his field-marshal role.
RUS_generate_kliment_voroshilov = {
	if = {
		limit = {
			has_character = RUS_kliment_voroshilov
			RUS_kliment_voroshilov = { is_unit_leader = no }
		}
		RUS_kliment_voroshilov = {
			add_field_marshal_role = {
				traits = { old_guard inflexible_strategist politically_connected }
				skill = 4
				attack_skill = 3
				defense_skill = 5
				planning_skill = 4
				logistics_skill = 6
			}
		}
	}
	if = {
		limit = {
			NOT = { has_country_flag = RUS_voroshilov_yeti_partisan_spawned }
		}
		set_country_flag = RUS_voroshilov_yeti_partisan_spawned
		if = {
			limit = {
				NOT = { has_template = "RUS_zhdanov_partisan_template" }
			}
			division_template = {
				name = "RUS_zhdanov_partisan_template"
				division_names_group = RUS_INF_01
				regiments = {
					infantry = { x = 0 y = 0 }
					infantry = { x = 0 y = 1 }
					infantry = { x = 0 y = 2 }
					infantry = { x = 1 y = 0 }
					infantry = { x = 1 y = 1 }
					infantry = { x = 1 y = 2 }
					infantry = { x = 2 y = 0 }
					artillery_brigade = { x = 2 y = 1 }
					artillery_brigade = { x = 2 y = 2 }
				}
			}
		}
		if = {
			limit = { controls_province = 9292 }
			572 = {
				create_unit = {
					division = "name = \"“雪怪”游击队\" division_template = \"RUS_zhdanov_partisan_template\" start_experience_factor = 0.6 start_equipment_factor = 1.0"
					prioritize_location = 9292
					owner = ROOT
				}
			}
		}
		else = {
			random_owned_controlled_state = {
				prioritize = { 572 }
				limit = { valid_state_for_spawning_friendly_unit = yes }
				create_unit = {
					division = "name = \"“雪怪”游击队\" division_template = \"RUS_zhdanov_partisan_template\" start_experience_factor = 0.6 start_equipment_factor = 1.0"
					owner = ROOT
				}
			}
		}
	}
}
