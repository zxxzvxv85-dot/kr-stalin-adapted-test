RUS_stalin_ensure_first_five_year_plan = {
	if = {
		limit = {
			has_country_flag = RUS_auto_kamenev_vst_left_path
		}
		if = {
			limit = { NOT = { has_variable = RUS_first_five_year_plan_score } }
			set_variable = { RUS_first_five_year_plan_score = 0 }
		}
		if = {
			limit = { NOT = { has_variable = RUS_first_five_year_plan_stage } }
			set_variable = { RUS_first_five_year_plan_stage = 0 }
		}
		if = {
			limit = { NOT = { has_variable = RUS_first_five_year_plan_consumer_goods_factor } }
			set_variable = { RUS_first_five_year_plan_consumer_goods_factor = 0 }
		}
		if = {
			limit = { NOT = { has_variable = RUS_first_five_year_plan_construction_speed } }
			set_variable = { RUS_first_five_year_plan_construction_speed = 0 }
		}
		if = {
			limit = { NOT = { has_variable = RUS_first_five_year_plan_efficiency_cap } }
			set_variable = { RUS_first_five_year_plan_efficiency_cap = 0 }
		}
		if = {
			limit = { NOT = { has_variable = RUS_first_five_year_plan_stability } }
			set_variable = { RUS_first_five_year_plan_stability = 0 }
		}
		if = {
			limit = { NOT = { has_variable = RUS_first_five_year_plan_resource_gain_efficiency } }
			set_variable = { RUS_first_five_year_plan_resource_gain_efficiency = 0 }
		}
		if = {
			limit = { NOT = { has_variable = RUS_first_five_year_plan_energy_from_coal } }
			set_variable = { RUS_first_five_year_plan_energy_from_coal = 0 }
		}
		if = {
			limit = { NOT = { has_variable = RUS_first_five_year_plan_factory_energy_consumption } }
			set_variable = { RUS_first_five_year_plan_factory_energy_consumption = 0 }
		}
		if = {
			limit = { NOT = { has_variable = RUS_first_five_year_plan_lack_of_resource_penalty_factor } }
			set_variable = { RUS_first_five_year_plan_lack_of_resource_penalty_factor = 0 }
		}
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = RUS_first_five_year_plan } } }
			add_dynamic_modifier = { modifier = RUS_first_five_year_plan }
		}
		if = {
			limit = {
				RUS_completed_focus_4_year_plan = yes
				NOT = { has_country_flag = RUS_first_five_year_plan_mission_started }
				NOT = { has_country_flag = RUS_first_five_year_plan_success }
				NOT = { has_country_flag = RUS_first_five_year_plan_failure }
			}
			set_country_flag = RUS_first_five_year_plan_mission_started
			activate_mission = RUS_first_five_year_plan_mission
		}
		RUS_stalin_refresh_first_five_year_plan_stage = yes
	}
}

RUS_stalin_refresh_first_five_year_plan_stage = {
	if = {
		limit = {
			has_country_flag = RUS_auto_kamenev_vst_left_path
			has_dynamic_modifier = { modifier = RUS_first_five_year_plan }
		}
		hidden_effect = {
			RUS_stalin_reset_first_five_year_plan_stage_values = yes
			if = {
				limit = { check_variable = { RUS_first_five_year_plan_score > 24.9 } }
				RUS_stalin_set_first_five_year_plan_stage_1 = yes
			}
			if = {
				limit = { check_variable = { RUS_first_five_year_plan_score > 49.9 } }
				RUS_stalin_set_first_five_year_plan_stage_2 = yes
			}
			if = {
				limit = { check_variable = { RUS_first_five_year_plan_score > 74.9 } }
				RUS_stalin_set_first_five_year_plan_stage_3 = yes
			}
			if = {
				limit = { check_variable = { RUS_first_five_year_plan_score > 99.9 } }
				RUS_stalin_set_first_five_year_plan_stage_4 = yes
			}
			if = {
				limit = { check_variable = { RUS_first_five_year_plan_score > 124.9 } }
				RUS_stalin_set_first_five_year_plan_stage_5 = yes
			}
			if = {
				limit = { check_variable = { RUS_first_five_year_plan_score > 149.9 } }
				RUS_stalin_set_first_five_year_plan_stage_6 = yes
			}
			if = {
				limit = {
					check_variable = { RUS_first_five_year_plan_stage > 1.9 }
					check_variable = { RUS_fpc_max_optimization_tier < 5 }
				}
				set_variable = { RUS_fpc_max_optimization_tier = 5 }
			}
			if = {
				limit = {
					check_variable = { RUS_first_five_year_plan_stage > 3.9 }
					check_variable = { RUS_fpc_max_optimization_tier < 6 }
				}
				set_variable = { RUS_fpc_max_optimization_tier = 6 }
			}
			if = {
				limit = {
					check_variable = { RUS_first_five_year_plan_stage > 5.9 }
					check_variable = { RUS_fpc_max_optimization_tier < 7 }
				}
				set_variable = { RUS_fpc_max_optimization_tier = 7 }
			}
			if = {
				limit = { has_country_flag = RUS_first_five_year_plan_success }
				add_to_variable = { RUS_first_five_year_plan_stability = 0.10 }
				add_to_variable = { RUS_first_five_year_plan_lack_of_resource_penalty_factor = -0.10 }
			}
			if = {
				limit = { has_country_flag = RUS_first_five_year_plan_failure }
				add_to_variable = { RUS_first_five_year_plan_stability = -0.10 }
				add_to_variable = { RUS_first_five_year_plan_lack_of_resource_penalty_factor = 0.10 }
			}
			if = {
				limit = {
					has_country_flag = RUS_first_five_year_plan_tankograd_completed
					check_variable = { RUS_first_five_year_plan_score > 99.9 }
					NOT = { has_country_flag = RUS_first_five_year_plan_tankograd_bonus }
				}
				set_country_flag = RUS_first_five_year_plan_tankograd_bonus
				add_ideas = RUS_first_five_year_plan_tankograd_bonus_hidden
			}
		}
	}
}

RUS_stalin_industry_effect = {
	if = {
		limit = {
			has_government = national_populist
			has_completed_focus = RUS_solidarist_representation
		}
		if = {
			limit = { has_stability < 0.75 }
			add_stability = 0.005
		}
		else = {
			add_popularity = { ideology = national_populist popularity = 0.005 }
		}
	}
	else_if = {
		limit = {
			OR = {
				RUS_completed_focus_4_year_plan = yes
				RUS_completed_focus_planning_directory = yes
			}
		}
		add_stability = -0.005
	}
	else_if = {
		limit = { has_completed_focus = RUS_new_russian_capitalism }
		add_stability = 0.005
	}
	if = {
		limit = { RUS_completed_focus_4_year_plan = yes }
		RUS_stalin_ensure_first_five_year_plan = yes
	}
}

		limit = { has_country_flag = RUS_universal_labour_service_industry_time_halved }
		multiply_variable = { RUS_ic_time_60 = 0.5 }
		multiply_variable = { RUS_ic_time_120 = 0.5 }
	}
	if = {
		limit = { has_country_flag = RUS_tesla_industrial_programme_active }
		multiply_variable = { RUS_ic_time_60 = 0.5 }
		multiply_variable = { RUS_ic_time_120 = 0.5 }
	}
}
