		limit = {
			check_variable = { RUS_fpc_focus_var = 0 }
		}

			text = {
				add_ideas = RUS_fpc_no_focus_optimization_[RUS_FPC_OPT]_idea

RUS_fpc_start = {
	if = {
		limit = { NOT = { has_country_flag = RUS_fpc_active } }
		set_country_flag = RUS_fpc_active
		set_variable = { RUS_fpc_optimization_tier = 2 }
		set_variable = { RUS_fpc_max_optimization_tier = 4 }
		set_variable = { RUS_fpc_focus_var = 0 }
		RUS_fpc_update_tier = yes
		mark_focus_tree_layout_dirty = yes
		custom_effect_tooltip = RUS_fpc_start_tt
	}
}
