RUS_stalin_enable_kamenev_bop = {
	if = {
		limit = {
			has_country_flag = RUS_auto_kamenev_vst_left_path
			NOT = { has_power_balance = { id = RUS_kamenev_power_balance } }
		}
		set_country_flag = RUS_kamenev_bop_active
		set_power_balance = {
			id = RUS_kamenev_power_balance
			left_side = RUS_kamenev_psr_side
			right_side = RUS_kamenev_bolshevik_side
			set_value = -0.45
		}
		RUS_stalin_kamenev_bop_refresh_runtime_effects = yes
		country_event = { id = stalin_form_sov.24 hours = 1 }
	}
}

RUS_stalin_kamenev_bop_ensure_active = {
	if = {
		limit = {
			OR = {
				has_country_flag = RUS_auto_kamenev_vst_left_path
				has_country_flag = RUS_kamenev_bop_active
			}
			NOT = { has_power_balance = { id = RUS_kamenev_power_balance } }
		}
		set_country_flag = RUS_kamenev_bop_active
		set_power_balance = {
			id = RUS_kamenev_power_balance
			left_side = RUS_kamenev_psr_side
			right_side = RUS_kamenev_bolshevik_side
			set_value = -0.45
		}
		RUS_stalin_kamenev_bop_refresh_runtime_effects = yes
	}
}

RUS_stalin_kamenev_bop_refresh_runtime_effects = {
	RUS_stalin_sverdlov_refresh_sic_trait = yes
	RUS_stalin_kamenev_bop_refresh_vst_idea_modifier = yes
	if = {
		limit = { has_power_balance = { id = RUS_kamenev_power_balance } }
		RUS_stalin_kamenev_bop_swap_character_traits = yes
	}
}

RUS_stalin_sverdlov_refresh_sic_trait = {
	if = {
		limit = {
			RUS_yakov_sverdlov = {
				is_second_in_command = yes
				has_trait = KR_red_eminence
			}
		}
		RUS_yakov_sverdlov = {
			remove_trait = { trait = KR_red_eminence slot = second_in_command }
			add_trait = { trait = RUS_red_eminence_sverdlov_sic slot = second_in_command }
		}
	}
}

RUS_stalin_kamenev_bop_refresh_vst_idea_modifier = {
	if = {
		limit = {
			has_power_balance = { id = RUS_kamenev_power_balance }
			OR = {
				has_idea = RUS_vst_party_unity_idea
				has_idea = RUS_vst_party_factionalism_idea
			}
			NOT = { has_dynamic_modifier = { modifier = RUS_kamenev_vst_party_unity_bop_modifier } }
		}
		add_dynamic_modifier = { modifier = RUS_kamenev_vst_party_unity_bop_modifier }
	}
	if = {
		limit = {
			has_dynamic_modifier = { modifier = RUS_kamenev_vst_party_unity_bop_modifier }
			OR = {
				NOT = { has_power_balance = { id = RUS_kamenev_power_balance } }
				NOT = {
					OR = {
						has_idea = RUS_vst_party_unity_idea
						has_idea = RUS_vst_party_factionalism_idea
					}
				}
			}
		}
		remove_dynamic_modifier = { modifier = RUS_kamenev_vst_party_unity_bop_modifier }
	}
}
