RUS_stalin_kamenev_migrate_legacy_balance = {
	if = {
		limit = { has_power_balance = { id = RUS_kamenev_power_balance } }
		set_country_flag = RUS_kamenev_bop_active
		if = {
			limit = { NOT = { power_balance_value = { id = RUS_kamenev_power_balance value < 0.75 } } }
			set_variable = { RUS_kamenev_party_balance = 87 }
		}
		else_if = {
			limit = { NOT = { power_balance_value = { id = RUS_kamenev_power_balance value < 0.50 } } }
			set_variable = { RUS_kamenev_party_balance = 62 }
		}
		else_if = {
			limit = { NOT = { power_balance_value = { id = RUS_kamenev_power_balance value < 0.25 } } }
			set_variable = { RUS_kamenev_party_balance = 37 }
		}
		else_if = {
			limit = { power_balance_value = { id = RUS_kamenev_power_balance value > -0.25 } }
			set_variable = { RUS_kamenev_party_balance = 0 }
		}
		else_if = {
			limit = { power_balance_value = { id = RUS_kamenev_power_balance value > -0.50 } }
			set_variable = { RUS_kamenev_party_balance = -37 }
		}
		else_if = {
			limit = { power_balance_value = { id = RUS_kamenev_power_balance value > -0.75 } }
			set_variable = { RUS_kamenev_party_balance = -62 }
		}
		else = {
			set_variable = { RUS_kamenev_party_balance = -87 }
		}
		remove_power_balance = { id = RUS_kamenev_power_balance }
		set_country_flag = RUS_kamenev_party_balance_migrated
		RUS_stalin_kamenev_balance_refresh = yes
	}
}

	multiply_variable = { RUS_kamenev_balance_gui_x = 1.8 }
	add_to_variable = { RUS_kamenev_balance_gui_x = 236 }
	clamp_variable = { var = RUS_kamenev_balance_gui_x min = 56 max = 416 }

	set_variable = { RUS_kamenev_balance_compliance_growth = 0 }
	set_variable = { RUS_kamenev_balance_stability_factor = 0 }
	set_variable = { RUS_kamenev_balance_research_speed_factor = 0 }

	set_variable = { RUS_kamenev_balance_pp_gain = 0 }

	set_variable = { RUS_kamenev_balance_efficiency_gain = 0 }

	set_variable = { RUS_kamenev_balance_building_slots = 0 }
	set_variable = { RUS_kamenev_balance_conscription_factor = 0 }

	if = {

		set_variable = { RUS_kamenev_balance_efficiency_gain = 0.30 }
		set_variable = { RUS_kamenev_balance_factory_output = 0.15 }
		set_variable = { RUS_kamenev_balance_pp_factor = 0.15 }
		set_variable = { RUS_kamenev_balance_building_slots = 0.20 }
		set_variable = { RUS_kamenev_balance_stability_factor = -0.15 }
		set_variable = { RUS_kamenev_balance_conscription_factor = -0.12 }

	else_if = {

		set_variable = { RUS_kamenev_balance_efficiency_gain = 0.20 }
		set_variable = { RUS_kamenev_balance_factory_output = 0.10 }

		set_variable = { RUS_kamenev_balance_conscription_factor = -0.08 }

	else_if = {

		set_variable = { RUS_kamenev_balance_efficiency_gain = 0.10 }
		set_variable = { RUS_kamenev_balance_factory_output = 0.05 }
		set_variable = { RUS_kamenev_balance_pp_factor = 0.05 }
		set_variable = { RUS_kamenev_balance_stability_factor = -0.05 }
		set_variable = { RUS_kamenev_balance_conscription_factor = -0.04 }
	}
	else_if = {
		limit = { check_variable = { RUS_kamenev_party_balance > -25 } }
		set_variable = { RUS_kamenev_balance_pp_gain = 0.05 }
		set_variable = { RUS_kamenev_balance_command_power_gain_mult = 0.10 }
	}
	else_if = {
		limit = { check_variable = { RUS_kamenev_party_balance > -50 } }
		set_variable = { RUS_kamenev_balance_stability_factor = 0.04 }
		set_variable = { RUS_kamenev_balance_research_speed_factor = 0.05 }
		set_variable = { RUS_kamenev_balance_pp_gain = -0.05 }
	}
	else_if = {
		limit = { check_variable = { RUS_kamenev_party_balance > -75 } }
		set_variable = { RUS_kamenev_balance_compliance_growth = 0.02 }
		set_variable = { RUS_kamenev_balance_stability_factor = 0.08 }
		set_variable = { RUS_kamenev_balance_research_speed_factor = 0.10 }
		set_variable = { RUS_kamenev_balance_build_speed_factor = 0.05 }
		set_variable = { RUS_kamenev_balance_pp_gain = -0.10 }
	}
	else = {
		set_variable = { RUS_kamenev_balance_compliance_growth = 0.04 }
		set_variable = { RUS_kamenev_balance_stability_factor = 0.12 }
		set_variable = { RUS_kamenev_balance_research_speed_factor = 0.10 }
		set_variable = { RUS_kamenev_balance_build_speed_factor = 0.10 }

RUS_stalin_psr_balance_collapse = {
	# The collapse is caused only by Socialist Revolutionary dominance (negative balance values).
	# `return` is not a valid effect, so the guards are inlined into a single if.
	if = {
		limit = {
			check_variable = { RUS_kamenev_party_balance < -74 }
			NOT = {
				has_country_flag = RUS_stalin_psr_balance_collapse_done
				has_dynamic_modifier = { modifier = RUS_stalin_psr_balance_collapse_modifier }
			}
		}
		set_country_flag = RUS_stalin_psr_balance_collapse_done
		add_dynamic_modifier = { modifier = RUS_stalin_psr_balance_collapse_modifier }
	}
}

	if = { limit = { has_idea = RUS_yevgeny_sattel_advisor } add_to_variable = { RUS_kamenev_weekly_balance_drift = -1 } }
	if = { limit = { has_idea = RUS_yakov_blyumkin_advisor } add_to_variable = { RUS_kamenev_weekly_balance_drift = -1 } }

	# Maximalist advisors counter whichever faction currently dominates.
	set_variable = { RUS_maximalist_advisor_centre_drift = 0 }
	if = { limit = { has_idea = RUS_aleksey_ustinov_advisor } add_to_variable = { RUS_maximalist_advisor_centre_drift = 1 } }
	if = { limit = { has_idea = RUS_andrey_kolegayev_advisor } add_to_variable = { RUS_maximalist_advisor_centre_drift = 1 } }

		multiply_variable = { RUS_kamenev_party_balance_abs = -1 }
	}
	if = {
		limit = { check_variable = { RUS_maximalist_advisor_centre_drift > RUS_kamenev_party_balance_abs } }

	}
	if = {
		limit = { check_variable = { RUS_kamenev_party_balance < 0 } }
		add_to_variable = { RUS_kamenev_weekly_balance_drift = RUS_maximalist_advisor_centre_drift }
	}
	else_if = {
		limit = { check_variable = { RUS_kamenev_party_balance > 0 } }
		multiply_variable = { RUS_maximalist_advisor_centre_drift = -1 }
		add_to_variable = { RUS_kamenev_weekly_balance_drift = RUS_maximalist_advisor_centre_drift }
	}

	if = { limit = { has_dynamic_modifier = { modifier = RUS_kamenev_psr_agenda_bop_modifier } } add_to_variable = { RUS_kamenev_weekly_balance_drift = -2 } }

	if = {
		limit = { has_country_flag = RUS_kamenev_party_balance_frozen }
		set_variable = { RUS_kamenev_weekly_balance_drift = 0 }
	}

	set_variable = { RUS_kamenev_weekly_balance_drift_abs = RUS_kamenev_weekly_balance_drift }
	if = {
		limit = { check_variable = { RUS_kamenev_weekly_balance_drift_abs < 0 } }
		multiply_variable = { RUS_kamenev_weekly_balance_drift_abs = -1 }
	}
}
