RUS_agri_set_base_rates = {
	if = {
		limit = { check_variable = { RUS_agri_season = 1 } }
		set_variable = { RUS_agri_wheat_base = 1.25 }
		set_variable = { RUS_agri_rye_base = 1.25 }
		set_variable = { RUS_agri_beet_base = 0.40 }
		set_variable = { RUS_agri_flax_base = 0.40 }
		set_variable = { RUS_agri_cotton_base = 0.40 }
	}
	else_if = {
		limit = { check_variable = { RUS_agri_season = 2 } }
		set_variable = { RUS_agri_wheat_base = 1.00 }
		set_variable = { RUS_agri_rye_base = 1.00 }
		set_variable = { RUS_agri_beet_base = 1.00 }
		set_variable = { RUS_agri_flax_base = 1.00 }
		set_variable = { RUS_agri_cotton_base = 1.25 }
	}
	else_if = {
		limit = { check_variable = { RUS_agri_season = 3 } }
		set_variable = { RUS_agri_wheat_base = 0.40 }
		set_variable = { RUS_agri_rye_base = 0.40 }
		set_variable = { RUS_agri_beet_base = 1.25 }
		set_variable = { RUS_agri_flax_base = 1.25 }
		set_variable = { RUS_agri_cotton_base = 1.25 }
	}
	else = {
		set_variable = { RUS_agri_wheat_base = 1.00 }
		set_variable = { RUS_agri_rye_base = 1.00 }
		set_variable = { RUS_agri_beet_base = 0.40 }
		set_variable = { RUS_agri_flax_base = 1.00 }
		set_variable = { RUS_agri_cotton_base = 1.00 }
	}

	if = {
		limit = { has_completed_focus = RUS_future_foreign_017 }
		add_to_variable = { RUS_agri_beet_base = 0.10 }
		add_to_variable = { RUS_agri_flax_base = 0.10 }
		add_to_variable = { RUS_agri_cotton_base = 0.10 }
	}
}

RUS_agri_refresh_totals = {
	set_variable = { RUS_agri_total_investment = RUS_agri_wheat_investment }
	add_to_variable = { RUS_agri_total_investment = RUS_agri_rye_investment }
	add_to_variable = { RUS_agri_total_investment = RUS_agri_beet_investment }
	add_to_variable = { RUS_agri_total_investment = RUS_agri_flax_investment }
	add_to_variable = { RUS_agri_total_investment = RUS_agri_cotton_investment }

	set_variable = {
		RUS_agri_unallocated_funds = {
			value = RUS_agri_capital
			subtract = RUS_agri_total_investment
			clamp = { min = 0 max = 30 }
		}
	}
	set_variable = {
		RUS_agri_investment_limit = {
			value = RUS_agri_capital
			round = yes
			clamp = { min = 0 max = 20 }
		}
	}
	if = {
		limit = { check_variable = { RUS_agri_investment_limit > RUS_agri_capital } }
		add_to_variable = { RUS_agri_investment_limit = -1 }
	}
	clamp_variable = { var = RUS_agri_investment_limit min = 0 max = 20 }

	clr_country_flag = RUS_agri_saturation_wheat
	clr_country_flag = RUS_agri_saturation_rye
	clr_country_flag = RUS_agri_saturation_beet
	clr_country_flag = RUS_agri_saturation_flax
	clr_country_flag = RUS_agri_saturation_cotton

	set_variable = { RUS_agri_concentration_check = RUS_agri_wheat_investment }
	multiply_variable = { RUS_agri_concentration_check = 2 }
	if = {
		limit = {
			has_country_flag = RUS_agri_previous_dominant_wheat
			check_variable = { RUS_agri_total_investment > 0 }
			check_variable = { RUS_agri_concentration_check > RUS_agri_total_investment }
		}
		set_country_flag = RUS_agri_saturation_wheat
	}
	set_variable = { RUS_agri_concentration_check = RUS_agri_rye_investment }
	multiply_variable = { RUS_agri_concentration_check = 2 }
	if = {
		limit = {
			has_country_flag = RUS_agri_previous_dominant_rye
			check_variable = { RUS_agri_total_investment > 0 }
			check_variable = { RUS_agri_concentration_check > RUS_agri_total_investment }
		}
		set_country_flag = RUS_agri_saturation_rye
	}
	set_variable = { RUS_agri_concentration_check = RUS_agri_beet_investment }
	multiply_variable = { RUS_agri_concentration_check = 2 }
	if = {
		limit = {
			has_country_flag = RUS_agri_previous_dominant_beet
			check_variable = { RUS_agri_total_investment > 0 }
			check_variable = { RUS_agri_concentration_check > RUS_agri_total_investment }
		}
		set_country_flag = RUS_agri_saturation_beet
	}
	set_variable = { RUS_agri_concentration_check = RUS_agri_flax_investment }
	multiply_variable = { RUS_agri_concentration_check = 2 }
	if = {
		limit = {
			has_country_flag = RUS_agri_previous_dominant_flax
			check_variable = { RUS_agri_total_investment > 0 }
			check_variable = { RUS_agri_concentration_check > RUS_agri_total_investment }
		}
		set_country_flag = RUS_agri_saturation_flax
	}
	set_variable = { RUS_agri_concentration_check = RUS_agri_cotton_investment }
	multiply_variable = { RUS_agri_concentration_check = 2 }
	if = {
		limit = {
			has_country_flag = RUS_agri_previous_dominant_cotton
			check_variable = { RUS_agri_total_investment > 0 }
			check_variable = { RUS_agri_concentration_check > RUS_agri_total_investment }
		}
		set_country_flag = RUS_agri_saturation_cotton
	}

	RUS_agri_refresh_gui = yes
}

	clr_country_flag = RUS_agri_annual_tier_bumper
	if = { limit = { has_country_flag = RUS_agri_emergency_advance_used } set_country_flag = RUS_agri_last_year_emergency }

	if = {
		limit = { check_variable = { RUS_agri_capital < 8 } }
		set_country_flag = RUS_agri_annual_tier_shortfall
		add_timed_idea = { idea = RUS_agri_annual_shortfall days = 365 }
	}
	else_if = {
		limit = { check_variable = { RUS_agri_capital < 12 } }
		set_country_flag = RUS_agri_annual_tier_balanced
	}
	else_if = {
		limit = {
			OR = {
				has_country_flag = RUS_agri_emergency_advance_used
				check_variable = { RUS_agri_capital < 16 }
			}
		}
		set_country_flag = RUS_agri_annual_tier_surplus
		add_timed_idea = { idea = RUS_agri_annual_surplus days = 365 }
	}
	else = {
		set_country_flag = RUS_agri_annual_tier_bumper
		add_timed_idea = { idea = RUS_agri_annual_bumper_surplus days = 365 }
	}
}

RUS_agri_settle_quarter = {
	if = {
		limit = { NOT = { has_country_flag = RUS_agri_allocation_locked } }
		RUS_agri_balance_allocation = yes
		set_country_flag = RUS_agri_allocation_locked
	}
	RUS_agri_set_base_rates = yes
	RUS_agri_refresh_totals = yes
	RUS_agri_calculate_final_rates = yes

	set_variable = { RUS_agri_wheat_return = RUS_agri_wheat_investment }
	multiply_variable = { RUS_agri_wheat_return = RUS_agri_wheat_final }
	set_variable = { RUS_agri_rye_return = RUS_agri_rye_investment }
	multiply_variable = { RUS_agri_rye_return = RUS_agri_rye_final }
	set_variable = { RUS_agri_beet_return = RUS_agri_beet_investment }
	multiply_variable = { RUS_agri_beet_return = RUS_agri_beet_final }
	set_variable = { RUS_agri_flax_return = RUS_agri_flax_investment }
	multiply_variable = { RUS_agri_flax_return = RUS_agri_flax_final }
	set_variable = { RUS_agri_cotton_return = RUS_agri_cotton_investment }
	multiply_variable = { RUS_agri_cotton_return = RUS_agri_cotton_final }

	set_variable = { RUS_agri_next_capital = RUS_agri_unallocated_funds }
	add_to_variable = { RUS_agri_next_capital = RUS_agri_wheat_return }
	add_to_variable = { RUS_agri_next_capital = RUS_agri_rye_return }
	add_to_variable = { RUS_agri_next_capital = RUS_agri_beet_return }
	add_to_variable = { RUS_agri_next_capital = RUS_agri_flax_return }
	add_to_variable = { RUS_agri_next_capital = RUS_agri_cotton_return }
	clamp_variable = { var = RUS_agri_next_capital min = 0 max = 30 }
	multiply_variable = { RUS_agri_next_capital = 10 }
	set_variable = { RUS_agri_next_capital = { value = RUS_agri_next_capital round = yes } }
	divide_variable = { RUS_agri_next_capital = 10 }
	set_variable = { RUS_agri_capital = RUS_agri_next_capital }
	set_variable = { RUS_agri_last_capital = RUS_agri_capital }
	set_variable = { RUS_agri_last_wheat_final = RUS_agri_wheat_final }
	set_variable = { RUS_agri_last_rye_final = RUS_agri_rye_final }
	set_variable = { RUS_agri_last_beet_final = RUS_agri_beet_final }
	set_variable = { RUS_agri_last_flax_final = RUS_agri_flax_final }
	set_variable = { RUS_agri_last_cotton_final = RUS_agri_cotton_final }
	set_country_flag = RUS_agri_has_previous_result
	RUS_agri_record_dominant_crop = yes

	if = {
		limit = { check_variable = { RUS_agri_season < 4 } }
		add_to_variable = { RUS_agri_season = 1 }
		RUS_agri_start_quarter = yes
	}
	else = {
		clr_country_flag = RUS_agri_quarter_active
		RUS_agri_apply_annual_result = yes
		country_event = RUS_agricultural_management.1
	}
}

RUS_agri_daily_update = {
	if = {
		limit = { NOT = { has_variable = RUS_agri_days_remaining } }
		RUS_agri_start_new_year = yes
	}
	else_if = {
		limit = { has_country_flag = RUS_agri_quarter_active }
		RUS_agri_set_base_rates = yes
		add_to_variable = { RUS_agri_days_remaining = -1 }
		if = {
			limit = { check_variable = { RUS_agri_days_remaining < 1 } }
			RUS_agri_settle_quarter = yes
		}
		else = { RUS_agri_refresh_gui = yes }
	}
}
