RUS_fr_apply_armoured_firepower_coordination_bonus = {
	if = {
		limit = { NOT = { has_variable = RUS_fr_armoured_firepower_bonus } }
		set_variable = { RUS_fr_armoured_firepower_bonus = 0 }
	}
	clamp_variable = { var = RUS_fr_armoured_firepower_bonus min = 0 max = 0.15 }
	set_temp_variable = { RUS_fr_armour_increment = party_popularity@communist }
	# Party popularity is stored as a 0-1 fraction. Multiplying it by 0.1
	# grants one tenth of the displayed Communist support: 9.02% becomes 0.902%.
	multiply_temp_variable = { RUS_fr_armour_increment = 0.1 }
	add_to_variable = { RUS_fr_armoured_firepower_bonus = RUS_fr_armour_increment }
	if = {
		limit = { check_variable = { RUS_fr_armoured_firepower_bonus > 0.15 } }
		set_temp_variable = { RUS_fr_armour_overflow = RUS_fr_armoured_firepower_bonus }
		subtract_from_variable = { var = RUS_fr_armour_overflow value = 0.15 }
		subtract_from_variable = { var = RUS_fr_armour_increment value = RUS_fr_armour_overflow }
		set_variable = { RUS_fr_armoured_firepower_bonus = 0.15 }
	}
	if = {
		limit = { check_variable = { RUS_fr_armour_increment > 0 } }
		add_to_variable = { RUS_fr_unified_armor_attack_factor = RUS_fr_armour_increment tooltip = army_armor_attack_factor_tt }
		add_to_variable = { RUS_fr_unified_armor_defence_factor = RUS_fr_armour_increment tooltip = army_armor_defence_factor_tt }
	}
	# Other armaments focuses provide a net 10% armour bonus, while this repeatable
	# decision may contribute at most another 15%; the final displayed ceiling is 25%.
	if = {
		limit = { check_variable = { RUS_fr_unified_armor_attack_factor > 0.25 } }
		set_variable = { RUS_fr_unified_armor_attack_factor = 0.25 }
	}
	if = {
		limit = { check_variable = { RUS_fr_unified_armor_defence_factor > 0.25 } }
		set_variable = { RUS_fr_unified_armor_defence_factor = 0.25 }
	}
	force_update_dynamic_modifier = yes
}

# effects, so an out-of-order script call can never restore a stronger
# version of the combined reform spirit.
