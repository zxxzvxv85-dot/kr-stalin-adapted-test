RUS_fr_complete_mobility_legacy_focuses = {
	# The old artillery and motorisation focuses are mutually exclusive. The new
	# reform tree keeps their compatibility state hidden while reproducing their
	# actual rewards through native effect tooltips.
	RUS_fr_show_mobility_legacy_rewards = yes
	hidden_effect = {
		if = {
			limit = {
				has_completed_focus = RUS_artillery_production
				NOT = { has_completed_focus = RUS_motorisation }
			}
			set_country_flag = RUS_fr_artillery_legacy_rewards_applied
			uncomplete_national_focus = {
				focus = RUS_artillery_production
				uncomplete_children = no
				refund_political_power = no
			}
		}
		if = {
			limit = { NOT = { has_completed_focus = RUS_motorisation } }
			complete_national_focus = RUS_motorisation
		}
		if = {
			limit = { NOT = { has_completed_focus = RUS_roads_of_courage } }
			complete_national_focus = RUS_roads_of_courage
		}
	}
	if = {
		limit = { has_idea = RUS_military_production_idea_3 }
		remove_ideas = RUS_military_production_idea_3
		if = {
			limit = { NOT = { has_idea = RUS_military_production_idea_4 } }
			add_ideas = RUS_military_production_idea_4
		}
	}
	add_ideas = RUS_fr_mobility_basic_result
	set_country_flag = RUS_fr_mobility_line_finished
}

RUS_fr_calculate_specialisation_bonus = {
	set_variable = { RUS_fr_socialist_military_support = party_popularity@totalist }
	add_to_variable = { RUS_fr_socialist_military_support = party_popularity@communist }
	set_variable = { RUS_fr_specialist_primary_bonus = RUS_fr_socialist_military_support }
	multiply_variable = { RUS_fr_specialist_primary_bonus = 0.20 }
	clamp_variable = { var = RUS_fr_specialist_primary_bonus min = 0 max = 0.20 }
	set_variable = { RUS_fr_specialist_secondary_bonus = RUS_fr_specialist_primary_bonus }
	multiply_variable = { RUS_fr_specialist_secondary_bonus = 0.50 }
	set_variable = { RUS_fr_specialist_negative_bonus = RUS_fr_specialist_primary_bonus }
	multiply_variable = { RUS_fr_specialist_negative_bonus = -1 }
}

# The first two 120-day reform decisions use these monotonic completion
# effects, so an out-of-order script call can never restore a stronger
# version of the combined reform spirit.
RUS_fr_apply_rectification_stage_one = {
	if = {
		limit = { has_idea = RUS_fr_reform_disorganisation_4 }
		swap_ideas = {
			remove_idea = RUS_fr_reform_disorganisation_4
			add_idea = RUS_fr_reform_disorganisation_3
		}
	}
	set_country_flag = RUS_fr_reform_stage_1
}

RUS_fr_apply_rectification_stage_two = {
	if = {
		limit = { has_idea = RUS_fr_reform_disorganisation_4 }
		swap_ideas = {
			remove_idea = RUS_fr_reform_disorganisation_4
			add_idea = RUS_fr_reform_disorganisation_2
		}
	}
	else_if = {
		limit = { has_idea = RUS_fr_reform_disorganisation_3 }
		swap_ideas = {
			remove_idea = RUS_fr_reform_disorganisation_3
			add_idea = RUS_fr_reform_disorganisation_2
		}
	}
	set_country_flag = RUS_fr_reform_stage_1
	set_country_flag = RUS_fr_reform_stage_2
}
