			}
			RUS_fr_apply_comrades_abroad_army_reform_bonus = yes
		}
		hidden_effect = { remove_ideas = RUS_fr_command_merit_system_idea }

		# These ideas only cancel specialist penalties carried by the old
		# disorganisation spirit. Both sides disappear during consolidation.
		hidden_effect = {
			remove_ideas = {
				RUS_fr_equipment_standardisation_correction
				RUS_fr_armoured_production_correction
				RUS_fr_field_regulations_correction
				RUS_fr_climate_training_correction
				RUS_fr_core_operations_correction
				RUS_fr_red_academy_correction
				RUS_fr_armoured_coordination_correction
				RUS_fr_special_forces_correction
				RUS_fr_mobilisation_base_correction
			}
		}
		clr_country_flag = RUS_fr_officer_training_penalty
		set_country_flag = RUS_fr_officer_training_reformed
		every_army_leader = {
			limit = { has_unit_leader_trait = RUS_fr_fragmented_officer_training }
			remove_unit_leader_trait = RUS_fr_fragmented_officer_training
		}

		add_dynamic_modifier = { modifier = RUS_fr_unified_army_reform }
	}
}
