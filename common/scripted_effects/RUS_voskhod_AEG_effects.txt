# Country scope. Selects and persists the ten states used by the AEG decision.
RUS_select_AEG_nationalisation_states = {
	if = {
		limit = {
			OR = {
				NOT = { has_country_flag = RUS_aeg_nationalisation_states_selected }
				NOT = { has_variable = RUS_aeg_nationalisation_state_1 }
				NOT = { has_variable = RUS_aeg_nationalisation_state_2 }

				NOT = { has_variable = RUS_aeg_nationalisation_state_10 }
			}
		}

			limit = { has_state_flag = RUS_aeg_nationalisation_selected_state }
			clr_state_flag = RUS_aeg_nationalisation_selected_state
		}
		random_owned_controlled_state = {
			limit = {
				is_core_of = ROOT
				impassable = no
				infrastructure < 5
				energy_infrastructure < 1
				industrial_infrastructure < 1
				NOT = { has_state_flag = RUS_aeg_nationalisation_selected_state }
			}
			set_state_flag = RUS_aeg_nationalisation_selected_state
			save_global_event_target_as = RUS_aeg_nationalisation_state_1
			ROOT = { set_variable = { RUS_aeg_nationalisation_state_1 = PREV } }
		}
		random_owned_controlled_state = {
			limit = {
				is_core_of = ROOT

		}

			limit = {
				is_core_of = ROOT
				impassable = no

				energy_infrastructure < 1

				NOT = { has_state_flag = RUS_aeg_nationalisation_selected_state }

				energy_infrastructure < 1

		random_owned_controlled_state = {
			limit = {
				is_core_of = ROOT

		}

			limit = {
				is_core_of = ROOT

			limit = {
