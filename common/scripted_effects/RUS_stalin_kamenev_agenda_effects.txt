RUS_stalin_kamenev_ensure_party_agendas = {
	if = {
		limit = {
			has_country_flag = RUS_auto_kamenev_vst_left_path
			has_completed_focus = RUS_mensheviks
			NOT = { has_country_flag = RUS_kamenev_party_agendas_closed }
			NOT = { has_completed_focus = RUS_bolshevik_gamble }
		}
		if = {
			limit = {
				NOT = { has_country_flag = RUS_kamenev_psr_agenda_project_active }
				NOT = { has_completed_focus = RUS_kamenev_psr_final_party_congress }
			}
			RUS_stalin_kamenev_dispatch_psr_agenda = yes
		}
		if = {
			limit = {
				NOT = { has_country_flag = RUS_kamenev_bol_agenda_project_active }
				NOT = { has_completed_focus = RUS_kamenev_bol_undissolved_central_committee }
			}
			RUS_stalin_kamenev_dispatch_bol_agenda = yes
		}
	}
}
