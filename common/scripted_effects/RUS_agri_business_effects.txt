# Draw public business conditions only at quarter start; rewards are paid only by settlement.
RUS_agri_draw_business_conditions = {
	random_list = {
		1 = { set_variable = { RUS_agri_wheat_capacity = 3 } }
		1 = { set_variable = { RUS_agri_wheat_capacity = 4 } }
		1 = { set_variable = { RUS_agri_wheat_capacity = 5 } }
	}
	random_list = {
		1 = { set_variable = { RUS_agri_rye_capacity = 3 } }
		1 = { set_variable = { RUS_agri_rye_capacity = 4 } }
		1 = { set_variable = { RUS_agri_rye_capacity = 5 } }
	}
	random_list = {
		1 = { set_variable = { RUS_agri_beet_capacity = 3 } }
		1 = { set_variable = { RUS_agri_beet_capacity = 4 } }
		1 = { set_variable = { RUS_agri_beet_capacity = 5 } }
	}
	random_list = {
		1 = { set_variable = { RUS_agri_flax_capacity = 3 } }
		1 = { set_variable = { RUS_agri_flax_capacity = 4 } }
		1 = { set_variable = { RUS_agri_flax_capacity = 5 } }
	}
	random_list = {
		1 = { set_variable = { RUS_agri_cotton_capacity = 3 } }
		1 = { set_variable = { RUS_agri_cotton_capacity = 4 } }
		1 = { set_variable = { RUS_agri_cotton_capacity = 5 } }
	}
	set_variable = { RUS_agri_order_2_crop = 0 }
	set_variable = { RUS_agri_order_2_quantity = 0 }
	set_variable = { RUS_agri_order_2_cash = 0 }
	if = {
		limit = { check_variable = { RUS_agri_season = 1 } }
		random_list = {
			1 = { set_variable = { RUS_agri_order_1_crop = 3 } }
			1 = { set_variable = { RUS_agri_order_1_crop = 4 } }
			1 = { set_variable = { RUS_agri_order_1_crop = 5 } }
		}
	}
	else_if = {
		limit = { check_variable = { RUS_agri_season = 3 } }
		random_list = {
			1 = { set_variable = { RUS_agri_order_1_crop = 1 } }
			1 = { set_variable = { RUS_agri_order_1_crop = 2 } }
		}
	}
	else_if = {
		limit = { check_variable = { RUS_agri_season = 4 } }
		set_variable = { RUS_agri_order_1_crop = 3 }
	}
	else = {
		random_list = {
			1 = { set_variable = { RUS_agri_order_1_crop = 1 } }
			1 = { set_variable = { RUS_agri_order_1_crop = 2 } }
			1 = { set_variable = { RUS_agri_order_1_crop = 3 } }
			1 = { set_variable = { RUS_agri_order_1_crop = 4 } }
			1 = { set_variable = { RUS_agri_order_1_crop = 5 } }
		}
	}
	set_temp_variable = { RUS_agri_draw_second_order = 0 }
	if = {
		limit = { check_variable = { RUS_agri_quarter_investment_limit > 3 } }
		random_list = {
			40 = { set_temp_variable = { RUS_agri_draw_second_order = 0 } }
			60 = { set_temp_variable = { RUS_agri_draw_second_order = 1 } }
		}
	}
	if = {
		limit = { check_variable = { RUS_agri_draw_second_order = 1 } }
		if = {
			limit = { check_variable = { RUS_agri_order_1_crop = 1 } }
			random_list = {
				1 = { set_variable = { RUS_agri_order_2_crop = 2 } }
				1 = { set_variable = { RUS_agri_order_2_crop = 3 } }
				1 = { set_variable = { RUS_agri_order_2_crop = 4 } }
				1 = { set_variable = { RUS_agri_order_2_crop = 5 } }
			}
		}
		else_if = {
			limit = { check_variable = { RUS_agri_order_1_crop = 2 } }
			random_list = {
				1 = { set_variable = { RUS_agri_order_2_crop = 1 } }
				1 = { set_variable = { RUS_agri_order_2_crop = 3 } }
				1 = { set_variable = { RUS_agri_order_2_crop = 4 } }
				1 = { set_variable = { RUS_agri_order_2_crop = 5 } }
			}
		}
		else_if = {
			limit = { check_variable = { RUS_agri_order_1_crop = 3 } }
			random_list = {
				1 = { set_variable = { RUS_agri_order_2_crop = 1 } }
				1 = { set_variable = { RUS_agri_order_2_crop = 2 } }
				1 = { set_variable = { RUS_agri_order_2_crop = 4 } }
				1 = { set_variable = { RUS_agri_order_2_crop = 5 } }
			}
		}
		else_if = {
			limit = { check_variable = { RUS_agri_order_1_crop = 4 } }
			random_list = {
				1 = { set_variable = { RUS_agri_order_2_crop = 1 } }
				1 = { set_variable = { RUS_agri_order_2_crop = 2 } }
				1 = { set_variable = { RUS_agri_order_2_crop = 3 } }
				1 = { set_variable = { RUS_agri_order_2_crop = 5 } }
			}
		}
		else_if = {
			limit = { check_variable = { RUS_agri_order_1_crop = 5 } }
			random_list = {
				1 = { set_variable = { RUS_agri_order_2_crop = 1 } }
				1 = { set_variable = { RUS_agri_order_2_crop = 2 } }
				1 = { set_variable = { RUS_agri_order_2_crop = 3 } }
				1 = { set_variable = { RUS_agri_order_2_crop = 4 } }
			}
		}
	}
	# Both orders must fit even the smallest initial capital.
	set_temp_variable = { RUS_agri_order_limit_raw = RUS_agri_quarter_investment_limit }
	divide_temp_variable = { RUS_agri_order_limit_raw = 2 }
	set_temp_variable = { RUS_agri_order_limit = { value = RUS_agri_order_limit_raw round = yes } }
	if = {
		limit = { check_variable = { RUS_agri_order_limit > RUS_agri_order_limit_raw } }
		subtract_from_temp_variable = { RUS_agri_order_limit = 1 }
	}
	clamp_temp_variable = { var = RUS_agri_order_limit min = 1 max = 4 }
	if = {
		limit = { check_variable = { RUS_agri_order_1_crop > 0 } }
		random_list = {
			1 = { set_variable = { RUS_agri_order_1_quantity = 2 } }
			1 = { set_variable = { RUS_agri_order_1_quantity = 3 } }
			1 = { set_variable = { RUS_agri_order_1_quantity = 4 } }
		}
		if = {
			limit = { check_variable = { RUS_agri_order_1_quantity > RUS_agri_order_limit } }
			set_variable = { RUS_agri_order_1_quantity = RUS_agri_order_limit }
		}
		set_variable = { RUS_agri_order_1_cash = RUS_agri_order_1_quantity }
		multiply_variable = { RUS_agri_order_1_cash = 0.65 }
	}
	if = {
		limit = { check_variable = { RUS_agri_order_2_crop > 0 } }
		random_list = {
			1 = { set_variable = { RUS_agri_order_2_quantity = 2 } }
			1 = { set_variable = { RUS_agri_order_2_quantity = 3 } }
			1 = { set_variable = { RUS_agri_order_2_quantity = 4 } }
		}
		if = {
			limit = { check_variable = { RUS_agri_order_2_quantity > RUS_agri_order_limit } }
			set_variable = { RUS_agri_order_2_quantity = RUS_agri_order_limit }
		}
		set_variable = { RUS_agri_order_2_cash = RUS_agri_order_2_quantity }
		multiply_variable = { RUS_agri_order_2_cash = 0.65 }
	}
}

# Tier discounts apply to excess units, never retroactively to the whole crop.
RUS_agri_apply_capacity_rates = {
	set_variable = { RUS_agri_wheat_priced_return = RUS_agri_wheat_investment }
	multiply_variable = { RUS_agri_wheat_priced_return = RUS_agri_wheat_final }

		limit = {
			check_variable = { RUS_agri_wheat_capacity > 0 }
			check_variable = { RUS_agri_wheat_investment > RUS_agri_wheat_capacity }
		}
		set_temp_variable = { RUS_agri_capacity_excess = RUS_agri_wheat_investment }
		subtract_from_temp_variable = { RUS_agri_capacity_excess = RUS_agri_wheat_capacity }

		clamp_temp_variable = { var = RUS_agri_capacity_middle min = 0 max = 2 }
		subtract_from_temp_variable = { RUS_agri_capacity_excess = RUS_agri_capacity_middle }
		set_temp_variable = { RUS_agri_capacity_middle_rate = RUS_agri_wheat_final }

RUS_agri_settle_orders = {
	RUS_agri_refresh_business_display = yes
	set_variable = { RUS_agri_last_orders_completed = 0 }
	set_variable = { RUS_agri_last_order_cash = 0 }
	if = {
		limit = {
			check_variable = { RUS_agri_order_1_crop > 0 }
			check_variable = { RUS_agri_order_1_quantity > 0 }
			NOT = { check_variable = { RUS_agri_order_1_progress < RUS_agri_order_1_quantity } }
		}
		add_to_variable = { RUS_agri_last_orders_completed = 1 }
		add_to_variable = { RUS_agri_last_order_cash = RUS_agri_order_1_cash }
	}
