on_actions = {
	on_startup = {
		effect = {
			every_country = {
				limit = {
					original_tag = RUS
				}
				if = {
					limit = { has_completed_focus = RUS_found_sov_union }
					set_country_flag = RUS_soviet_foundation_super_event_seen
					clr_country_flag = RUS_soviet_foundation_super_event_armed
					clr_country_flag = RUS_soviet_foundation_playback_authorized
					clr_country_flag = RUS_soviet_foundation_playback_session_v2
					clr_country_flag = RUS_soviet_foundation_super_event_visible
					clr_country_flag = RUS_soviet_foundation_video_visible
				}
				else_if = {
					limit = { has_country_flag = RUS_soviet_foundation_video_visible }
					clr_country_flag = RUS_soviet_foundation_playback_authorized
					clr_country_flag = RUS_soviet_foundation_playback_session_v2
					clr_country_flag = RUS_soviet_foundation_video_visible
				}
				if = {
					limit = { is_ai = no }
					set_variable_to_random = RUS_soviet_foundation_super_event_refresh
				}
			}
		}
	}
}
