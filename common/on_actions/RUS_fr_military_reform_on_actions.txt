on_actions = {
	on_startup = {
		effect = {
			every_country = {
				limit = { original_tag = RUS }
				RUS_fr_update_timoshenko_decision_costs = yes
				RUS_fr_update_dashboard_reform_progress = yes
				RUS_fr_repair_comrades_abroad_army_reform_bonus = yes
				RUS_fr_repair_revolutionary_collectivism_conscription_bonus = yes
				RUS_fr_repair_light_infantry_special_forces_bonus = yes
				if = {
					limit = { has_country_flag = RUS_fr_military_reform_unlocked }
					RUS_fr_refresh_military_reform_materials = yes
				}
				country_event = { id = RUS_fr_military_reform.900 hours = 1 }
			}
		}
	}

	on_daily = {
		effect = {
			if = {
				limit = { tag = RUS }
				RUS_fr_update_timoshenko_decision_costs = yes
				RUS_fr_update_dashboard_reform_progress = yes
				RUS_fr_update_okmo_half_xp_cost = yes
				RUS_fr_repair_comrades_abroad_army_reform_bonus = yes
				RUS_fr_repair_revolutionary_collectivism_conscription_bonus = yes
				RUS_fr_repair_light_infantry_special_forces_bonus = yes
				if = {
					limit = { has_country_flag = RUS_fr_military_reform_unlocked }
					RUS_fr_add_daily_military_reform_materials = yes
				if = {
					limit = {
						has_country_flag = RUS_fr_military_reform_unlocked
						OR = {
							has_idea = RUS_white_army_red_baron
							has_idea = RUS_lack_of_officers
							has_idea = RUS_lack_of_officers_2
							has_idea = RUS_lack_of_officers_3
						}
					}
					remove_ideas = { RUS_white_army_red_baron RUS_lack_of_officers RUS_lack_of_officers_2 RUS_lack_of_officers_3 }
					if = {
						limit = {
							NOT = {
								OR = {
								has_idea = RUS_fr_reform_disorganisation_4
								has_idea = RUS_fr_reform_disorganisation_3
								has_idea = RUS_fr_reform_disorganisation_2
								has_idea = RUS_fr_reform_disorganisation_1
								has_idea = RUS_fr_reform_disorganisation_0
								has_idea = RUS_fr_reform_disorganisation_4_comrades_abroad
								has_idea = RUS_fr_reform_disorganisation_3_comrades_abroad
								has_idea = RUS_fr_reform_disorganisation_2_comrades_abroad
								has_idea = RUS_fr_reform_disorganisation_1_comrades_abroad
								has_idea = RUS_fr_reform_disorganisation_0_comrades_abroad
								}
							}
						}
						RUS_fr_add_initial_reform_disorganisation = yes
					}
				}
				}
			}
		}
	}

	on_weekly = {
		effect = {
			if = {
				limit = { tag = RUS }
				country_event = { id = RUS_fr_military_reform.900 hours = 1 }
			}
		}
	}

	# Cadres Decide Everything: 25% chance to gain +1 Attack on level-up.
	# Do not stack with the vanilla Bold Attack spirit, which has its own roll.
	on_unit_leader_level_up = {
		effect = {
			# Timoshenko's skill sets the reform discount; refresh it for RUS on level-up.
			# is_character is not used here on purpose: probing it inside `character = {}` logs an
			# error for every foreign leader that levels up, on top of being redundant.
			FROM = {
				if = {
					limit = { tag = RUS }
					RUS_fr_update_timoshenko_decision_costs = yes
				}
			}
			if = {
				limit = {
					FROM = {
						has_country_flag = RUS_fr_red_commanders_upgrade_bonus
						NOT = { has_idea = bold_attack_spirit }
					}
				}
				character = {
					if = {
						limit = { is_army_leader = yes }
						random_list = { 25 = { add_attack = 1 } 75 = { } }
					}
				}
			}
			if = {
				limit = {
					FROM = {
						has_country_flag = RUS_fr_specialise_armaments
						has_idea = RUS_semyon_timoshenko_theorist
					}
				}
				character = {
					if = {
						limit = { is_army_leader = yes }
						random_list = { 50 = { add_attack = 1 } 50 = { } }
						random_list = { 50 = { add_defense = 1 } 50 = { } }
						random_list = { 50 = { add_planning = 1 } 50 = { } }
						random_list = { 50 = { add_logistics = 1 } 50 = { } }
					}
				}
			}
		}
	}

	on_unit_leader_created = {
		effect = {
			character = {
				if = {
					limit = {
						FROM = { has_country_flag = RUS_fr_officer_training_penalty }
						is_army_leader = yes
					}
					root = {
						if = {
							limit = { NOT = { has_trait = RUS_fr_fragmented_officer_training } }
							add_unit_leader_trait = RUS_fr_fragmented_officer_training
						}
					}
				}
			}
		}
	}
}
